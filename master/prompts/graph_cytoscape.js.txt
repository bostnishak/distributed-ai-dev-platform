const data = JSON.parse(document.getElementById('data').textContent);
const graph = cytoscape({
  container: document.getElementById('view'),
  elements: data.elements,
  layout: { name: 'breadthfirst', directed: true, spacingFactor: 1.3 },
  style: [
    { selector: 'node', style: { label: 'data(label)', 'text-wrap': 'wrap', 'text-max-width': '150px', 'text-valign': 'center', 'text-halign': 'center', shape: 'round-rectangle', width: 170, height: 56, 'background-color': '#e8eefc', 'border-width': 2, 'border-color': '#3b5bdb', 'font-size': 13 } },
    { selector: 'node[kind = "decision"]', style: { shape: 'diamond', height: 96, 'border-color': '#d9480f' } },
    { selector: 'node[kind = "start"], node[kind = "end"]', style: { shape: 'ellipse' } },
    { selector: 'edge', style: { label: 'data(label)', 'curve-style': 'bezier', 'target-arrow-shape': 'triangle', width: 2, 'line-color': '#868e96', 'target-arrow-color': '#868e96', 'font-size': 12, 'text-background-color': '#ffffff', 'text-background-opacity': 1 } },
    { selector: '.faded', style: { opacity: 0.2 } }
  ]
});
graph.on('tap', 'node', (event) => {
  graph.elements().addClass('faded');
  event.target.closedNeighborhood().removeClass('faded');
});
graph.on('tap', (event) => { if (event.target === graph) { graph.elements().removeClass('faded'); } });
