"""Starts the application, waits until it answers and opens it in the browser. Used by run.ps1 and run_windows.bat."""

import argparse
import subprocess
import sys
import time
import urllib.request
import webbrowser

COMMAND = __COMMAND__

parser = argparse.ArgumentParser(description="Uygulamayı başlatır")
parser.add_argument("--port", type=int, default=__PORT__, help="dinlenecek port")
parser.add_argument("--no-browser", action="store_true", help="tarayıcıyı açma")
options = parser.parse_args()
url = f"http://127.0.0.1:{options.port}"
server = subprocess.Popen([sys.executable, *[part.replace("{port}", str(options.port)) for part in COMMAND]])
try:
    for _ in range(120):
        if server.poll() is not None:
            sys.exit(f"Sunucu açılırken durdu (kod {server.returncode}).")
        try:
            urllib.request.urlopen(url + "/", timeout=2)
            break
        except urllib.error.HTTPError:
            break  # it answers, even if this path is not a page
        except OSError:
            time.sleep(0.5)
    else:
        sys.exit("Sunucu 60 saniyede yanıt vermedi.")
    print(f"HAZIR {url}", flush=True)
    if not options.no_browser:
        webbrowser.open(url)
    server.wait()
finally:
    if server.poll() is None:
        server.terminate()
