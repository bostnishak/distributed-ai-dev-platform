// Loads /sandbox/index.html in jsdom, runs its scripts, clicks every button / link-like control and types into inputs.
// Prints one JSON line: {"errors": [...], "controls": N, "changed": M, "texts": [...]}
const { JSDOM, VirtualConsole } = require('jsdom');
const fs = require('fs');
const errors = [];
const vc = new VirtualConsole();
vc.on('jsdomError', (e) => /Could not parse CSS|Not implemented:/.test(String(e && e.message || e)) ? null : errors.push(String((e && (e.detail && e.detail.stack || e.stack)) || e).split('\n').slice(0, 3).join(' | ')));
vc.on('error', (...a) => errors.push('console.error: ' + a.map(String).join(' ').slice(0, 300)));
const memory = (w) => {
  let data = {};
  const store = { getItem: (k) => (k in data ? data[k] : null), setItem: (k, v) => { data[k] = String(v); }, removeItem: (k) => { delete data[k]; }, clear: () => { data = {}; } };
  for (const name of ['localStorage', 'sessionStorage']) Object.defineProperty(w, name, { configurable: true, value: store });
  w.confirm = () => true; w.alert = () => {};
};
const html = fs.readFileSync('/sandbox/index.html', 'utf8');
if (/class="hl-(tag|attr|str|kw|num|comment)"/.test(html)) errors.push('Kaynak dosyada editor renklendirme etiketleri (class="hl-...") var; saf kaynak kodu yaz.');
const stub = html.match(/lorem ipsum|TODO:|FIXME|mockData|fakeApi/i);
if (stub) errors.push('Sahte veri / yarim birakilmis yer tutucu var: "' + stub[0] + '". Gercek calisan kodla degistir.');
const MARK = 'Kalite123';
const wait = (ms) => new Promise((r) => setTimeout(r, ms));
// The core promise of a data app: a valid entry shows up on screen and the "All" filter keeps it there.
async function checkAdd(w, snapshot) {
  const d = w.document;
  const add = [...d.querySelectorAll('button, input[type=submit], input[type=button]')].find((b) => /(^|\s)(ekle|add|kaydet|save|oluştur|create)/i.test((b.textContent || b.value || '').trim()));
  const text = [...d.querySelectorAll('input:not([type]), input[type=text], textarea')].find((i) => !/ara|search|filtre|filter/i.test(i.id + i.placeholder));
  if (!add || !text) return;
  const scope = add.closest('form') || d;
  for (const inp of scope.querySelectorAll('input, textarea')) {
    if (/button|submit|checkbox|radio|search|file/.test(inp.type)) continue;
    inp.value = inp.type === 'number' ? '5' : inp.type === 'date' ? '2026-01-15' : inp.type === 'time' ? '10:00' : inp.type === 'email' ? 'a@b.co' : MARK;
    inp.dispatchEvent(new w.Event('input', { bubbles: true })); inp.dispatchEvent(new w.Event('change', { bubbles: true }));
  }
  text.value = MARK;
  for (const sel of scope.querySelectorAll('select')) { const o = [...sel.options].find((x) => x.value !== '' && !x.disabled); if (o) sel.value = o.value; }
  add.dispatchEvent(new w.MouseEvent('click', { bubbles: true, cancelable: true }));
  await wait(300);
  if (!snapshot().includes(MARK)) { errors.push('Ekleme calismiyor: form gecerli degerlerle dolduruldu ve "' + (add.textContent || add.value).trim() + '" tiklandi, ama girilen "' + MARK + '" ekranda gorunmuyor. Ekleme sonrasi liste yeniden cizilmeli ve secili filtre yeni kaydi gizlememeli.'); return; }
  const all = [...d.querySelectorAll('button')].find((b) => /^(tümü|hepsi|all)$/i.test(b.textContent.trim()));
  if (!all) return;
  all.dispatchEvent(new w.MouseEvent('click', { bubbles: true, cancelable: true }));
  await wait(200);
  if (!snapshot().includes(MARK)) errors.push('"' + all.textContent.trim() + '" filtresi kayitlari gizliyor: bu filtre secilince tum kayitlar gorunmeli.');
}
JSDOM.fromFile('/sandbox/index.html', { runScripts: 'dangerously', resources: 'usable', pretendToBeVisual: true, virtualConsole: vc, url: 'file:///sandbox/index.html', beforeParse: memory }).then((dom) => {
  const w = dom.window;
  w.addEventListener('error', (e) => errors.push('window.onerror: ' + e.message + ' @' + (e.filename || '') + ':' + (e.lineno || '')));
  setTimeout(async () => {
    const d = w.document;
    const snapshot = () => d.body.textContent.replace(/\s+/g, ' ').trim();
    const shown = () => { const c = d.body.cloneNode(true); c.querySelectorAll('script, style, [hidden]').forEach((n) => n.remove()); return c.textContent; };
    await checkAdd(w, shown);
    const before = snapshot();
    const controls = [...d.querySelectorAll('button, [onclick], input[type=button], input[type=submit], a[href^="#"], [role=button], .btn, .key, .keypad *')]
      .filter((el, i, arr) => arr.indexOf(el) === i && el.children.length < 3);
    let clicked = 0;
    for (const el of controls.slice(0, 80)) {
      try { el.dispatchEvent(new w.MouseEvent('click', { bubbles: true, cancelable: true })); clicked++; } catch (e) { errors.push('click failed: ' + e.message); }
    }
    for (const inp of [...d.querySelectorAll('input:not([type=button]):not([type=submit]):not([type=checkbox]):not([type=radio]), textarea')].slice(0, 12)) {
      try {
        inp.value = inp.type === 'number' ? '5' : inp.type === 'date' ? '2026-01-15' : inp.type === 'email' ? 'a@b.co' : 'Deneme';
        inp.dispatchEvent(new w.Event('input', { bubbles: true })); inp.dispatchEvent(new w.Event('change', { bubbles: true }));
      } catch (e) { errors.push('input failed: ' + e.message); }
    }
    for (const f of [...d.querySelectorAll('form')].slice(0, 4)) {
      try { f.dispatchEvent(new w.Event('submit', { bubbles: true, cancelable: true })); } catch (e) { errors.push('submit failed: ' + e.message); }
    }
    setTimeout(() => {
      const after = snapshot();
      console.log(JSON.stringify({ errors: [...new Set(errors)].slice(0, 8), controls: controls.length, clicked, changed: before !== after, title: d.title, texts: [after.slice(0, 160)] }));
      process.exit(0);
    }, 300);
  }, 700);
}).catch((e) => { console.log(JSON.stringify({ errors: ['load failed: ' + e.message], controls: 0, clicked: 0, changed: false })); process.exit(0); });
