<script>
const data = JSON.parse(document.getElementById('data').textContent);
// money is computed in whole numbers (BigInt), so no binary floating point rounding can change a digit
function scaled(text) { const [whole, part = ''] = text.split('.'); return [BigInt(whole + part), part.length]; }
function halfUp(numerator, denominator) { return (numerator * 2n + denominator) / (denominator * 2n); }
function cents(quantity, price) { const [q, qd] = scaled(quantity), [p, pd] = scaled(price); return halfUp(q * p * 100n, 10n ** BigInt(qd + pd)); }
function show(value) { const text = value.toString().padStart(3, '0'); return text.slice(0, -2).replace(/\B(?=(\d{3})+(?!\d))/g, '.') + ',' + text.slice(-2); }
const lines = data.items.map(item => ({...item, total: cents(item.quantity, item.price)}));
const subtotal = lines.reduce((sum, line) => sum + line.total, 0n);
const vat = halfUp(subtotal * BigInt(data.vat), 100n);
const largest = lines.reduce((top, line) => line.total > top ? line.total : top, 0n);
const summary = {'Ara toplam': subtotal, 'KDV': vat, 'Genel toplam': subtotal + vat, 'Ortalama kalem tutarı': halfUp(subtotal, BigInt(lines.length)), 'En yüksek kalem tutarı': largest};
document.getElementById('title').textContent = data.title;
for (const line of lines) {
  const row = document.createElement('tr');
  const cells = [line.name, line.quantity.replace('.', ','), line.price.replace('.', ','), show(line.total)];
  cells.forEach((text, index) => { const cell = document.createElement('td'); cell.textContent = text; if (index) cell.className = 'n'; if (index === 3) cell.dataset.figure = line.name; row.append(cell); });
  const share = document.createElement('td'); const bar = document.createElement('span'); bar.className = 'bar';
  bar.style.width = Number(largest ? line.total * 120n / largest : 0n) + 'px'; share.append(bar); row.append(share);
  document.getElementById('rows').append(row);
}
for (const [name, value] of Object.entries(summary)) {
  const term = document.createElement('dt'); term.textContent = name;
  const amount = document.createElement('dd'); amount.textContent = show(value); amount.dataset.figure = name;
  document.getElementById('summary').append(term, amount);
}
</script>
