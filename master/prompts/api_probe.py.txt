def _sample(schema, defs):
    if "$ref" in schema:
        return _sample(defs.get(schema["$ref"].rsplit("/", 1)[-1], {}), defs)
    if "anyOf" in schema:
        return _sample(schema["anyOf"][0], defs)
    if "enum" in schema:
        return schema["enum"][0]
    kind = schema.get("type")
    if kind == "object" or "properties" in schema:
        return {k: _sample(v, defs) for k, v in schema.get("properties", {}).items()}
    if kind == "array":
        return [_sample(schema.get("items", {}), defs)]
    fmt = {"date": "2026-01-15", "date-time": "2026-01-15T10:00:00", "email": "a@b.co"}.get(schema.get("format"))
    return fmt or {"integer": 1, "number": 1.5, "boolean": True}.get(kind, "deneme")


_HOSTILE = ("", "   ", "x" * 20000, "'; DROP TABLE t;--", "<script>alert(1)</script>", "ğüşİö 😀", -1, 0, 10**18, 1.5, None, [], {})


def _fuzz(client, method, path, body, params):
    """Boundary and hostile values, one field at a time: the app may refuse them (4xx) but must not crash."""
    if not isinstance(body, dict):
        return 0
    sent = 0
    for field in list(body)[:6]:
        for value in _HOSTILE:
            got = client.request(method, path, json={**body, field: value}, params=params)
            sent += 1
            if got.status_code >= 500:
                raise RuntimeError("%s %s returned HTTP %s for field %r = %.40r" % (method, path, got.status_code, field, value))
    broken = client.request(method, path, content=b"{not json", params=params, headers={"content-type": "application/json"})
    if broken.status_code >= 500:
        raise RuntimeError("%s %s returned HTTP %s for malformed JSON" % (method, path, broken.status_code))
    return sent + 1


def _check_roundtrip(client, doc, defs):
    """State must follow the requests: a created record is listed, a deleted one is gone."""
    paths = doc.get("paths", {})
    for path, ops in paths.items():
        if "{" in path or "post" not in ops or "get" not in ops:
            continue
        ref = ops["post"].get("requestBody", {}).get("content", {}).get("application/json", {}).get("schema")
        if not ref or any(p.get("required") for p in ops["get"].get("parameters", [])):
            continue
        schema = defs.get(ref.get("$ref", "").rsplit("/", 1)[-1], ref)
        body = _sample(schema, defs)
        text_fields = [k for k, v in schema.get("properties", {}).items() if v.get("type") == "string" and not v.get("format") and not v.get("enum") and not v.get("pattern")]
        if not text_fields or not isinstance(body, dict):
            continue
        marker = "iz7q" + str(len(path))
        body[text_fields[0]] = marker
        made = client.post(path, json=body)
        if made.status_code >= 300:
            continue
        listed = client.get(path)
        if listed.status_code == 200 and marker not in listed.text:
            raise RuntimeError("GET %s does not list the record that POST just created (not persisted?)" % path)
        record = made.json() if made.headers.get("content-type", "").startswith("application/json") else None
        key = record.get("id") if isinstance(record, dict) else None
        item = next((p for p in paths if p.startswith(path.rstrip("/") + "/{") and p.count("{") == 1 and "delete" in paths[p]), None)
        if key is not None and item:
            gone = client.delete(item[: item.index("{")] + str(key))
            if gone.status_code < 300 and marker in client.get(path).text:
                raise RuntimeError("DELETE %s reported success but GET %s still lists the record" % (item, path))
        print("API roundtrip %s ok" % path)


def check_fastapi(app):
    """Real requests against every path without parameters: valid body never 5xx, empty body of a required model is 4xx."""
    from fastapi.testclient import TestClient

    doc = app.openapi()
    defs = doc.get("components", {}).get("schemas", {})
    with TestClient(app, raise_server_exceptions=True) as client:
        for path, ops in doc.get("paths", {}).items():
            if "{" in path:
                continue
            for method, op in ops.items():
                body = op.get("requestBody", {}).get("content", {}).get("application/json", {}).get("schema")
                if method not in ("get", "post"):
                    continue
                params = {p["name"]: _sample(p.get("schema", {}), defs) for p in op.get("parameters", []) if p.get("required") and p.get("in") == "query"}
                if body:
                    required = defs.get(body.get("$ref", "").rsplit("/", 1)[-1], body).get("required")
                    if required:
                        bad = client.request(method.upper(), path, json={}, params=params)
                        if not 400 <= bad.status_code < 500:
                            raise RuntimeError("%s %s accepted an empty body (HTTP %s)" % (method.upper(), path, bad.status_code))
                    got = client.request(method.upper(), path, json=_sample(body, defs), params=params)
                    schema = defs.get(body.get("$ref", "").rsplit("/", 1)[-1], body)
                    print("API fuzz %s %s: %s requests" % (method.upper(), path, _fuzz(client, method.upper(), path, _sample(schema, defs), params)))
                else:
                    got = client.request(method.upper(), path, params=params)
                if got.status_code >= 500:
                    raise RuntimeError("%s %s returned HTTP %s" % (method.upper(), path, got.status_code))
                print("API %s %s -> %s" % (method.upper(), path, got.status_code))
        _check_roundtrip(client, doc, defs)


def check_flask_posts(app):
    client = app.test_client()
    for rule in app.url_map.iter_rules():
        if "POST" in rule.methods and "<" not in rule.rule:
            code = client.post(rule.rule, json={}).status_code
            if code >= 500:
                raise RuntimeError("POST %s with an empty body returned HTTP %s" % (rule.rule, code))
